Prisma — Video model

prisma/schema.prisma

model Video {
  id                   String    @id @default(cuid())
  url                  String
  caption              String
  description          String?
  thumbnailUrl         String?
  videoStoragePath     String?
  thumbnailStoragePath String?
  userId               String
  user                 User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  createdAt            DateTime  @default(now())
  likes                Like[]
  comments             Comment[]
}
